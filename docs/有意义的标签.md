// PredefinedLabel.dart - 使用自然键(英文名)作为ID的标签类
class PredefinedLabel {
  // 使用标签的英文名作为ID (例如 "spam", "friend", "work" 等)
  final String id;
  
  // 标签的显示文本 (可以是任何语言，例如中文 "垃圾", "朋友", "工作" 等)
  final String displayText;
  
  // 其他属性
  final String? avatar;
  final String? icon;
  
  const PredefinedLabel({
    required this.id,      // 例如 "spam"
    required this.displayText, // 例如 "垃圾"
    this.avatar,
    this.icon,
  });
  
  // 从英文名创建标签的工厂方法
  factory PredefinedLabel.fromEnglishName(
    String englishName, {
    String? displayText,
    String? avatar,
    String? icon,
  }) {
    // 将英文名转换为小写并去除空格，确保ID的规范性
    final normalizedId = englishName.toLowerCase().replaceAll(' ', '_');
    
    return PredefinedLabel(
      id: normalizedId,
      // 如果没有提供显示文本，就使用英文名
      displayText: displayText ?? englishName,
      avatar: avatar,
      icon: icon,
    );
  }
  
  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'displayText': displayText,
      'avatar': avatar,
      'icon': icon,
    };
  }
  
  factory PredefinedLabel.fromMap(Map<String, dynamic> map) {
    return PredefinedLabel(
      id: map['id'],
      displayText: map['displayText'],
      avatar: map['avatar'],
      icon: map['icon'],
    );
  }
}

// 预定义标签服务
class PredefinedLabelService {
  // 内部存储预定义的标签
  final Map<String, Label> _labels = {};
  
  // 初始化一些常用标签
  PredefinedLabelService() {
    // 添加一些预定义标签
    _addPredefinedLabels();
  }
  
  void _addPredefinedLabels() {
    // 添加一些常用标签
    final commonLabels = [
      Label.fromEnglishName('spam', displayText: '垃圾'),
      Label.fromEnglishName('friend', displayText: '朋友'),
      Label.fromEnglishName('work', displayText: '工作'),
      Label.fromEnglishName('family', displayText: '家人'),
      Label.fromEnglishName('important', displayText: '重要'),
      // 可以添加更多预定义标签
    ];
    
    // 将标签添加到内部存储
    for (final label in commonLabels) {
      _labels[label.id] = label;
    }
  }
  
  // 获取所有预定义标签
  List<Label> getAllLabels() {
    return _labels.values.toList();
  }
  
  // 通过ID获取标签
  Label? getLabelById(String id) {
    return _labels[id];
  }
  
  // 添加自定义标签
  Label addCustomLabel(String englishName, String displayText, {String? avatar, String? icon}) {
    // 创建新标签
    final label = Label.fromEnglishName(
      englishName,
      displayText: displayText,
      avatar: avatar,
      icon: icon,
    );
    
    // 存储标签
    _labels[label.id] = label;
    
    return label;
  }
  
  // 更新标签
  void updateLabel(Label label) {
    _labels[label.id] = label;
  }
  
  // 删除标签
  void removeLabel(String id) {
    _labels.remove(id);
  }
}
